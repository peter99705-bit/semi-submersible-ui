#include #include #include #include #include #include // ==================== 1. 硬體腳位定義 (對應 8路繼電器) ==================== const int RELAY_PINS[8] = {23, 22, 21, 19, 18, 5, 17, 16}; /* PIN 23 (IN 1): 立柱 1 微型打氣機 PIN 22 (IN 2): 立柱 1 0520D 閥門 (排氣) PIN 21 (IN 3): 立柱 2 微型打氣機 PIN 19 (IN 4): 立柱 2 0520D 閥門 (排氣) PIN 18 (IN 5): 立柱 3 微型打氣機 PIN 5 (IN 6): 立柱 3 0520D 閥門 (排氣) PIN 17 (IN 7): 立柱 4 微型打氣機 PIN 16 (IN 8): 立柱 4 0520D 閥門 (排氣) */ // ==================== 2. 藍牙 BLE 設定 ==================== #define WEB_SERVICE_UUID "4fafc201-1fb5-459e-8fcc-c5c9c331914b" #define WEB_CHAR_RX_UUID "beb5483e-36e1-4688-b7f5-ea07361b26a6" #define WEB_CHAR_TX_UUID "1b950702-f94f-438d-932d-92c90299fde2" // 您的 Rabboni 藍牙 MAC 位址 static BLEAddress rabboniAddress("E4:38:6A:16:6D:64"); static boolean rabboniConnected = false; static BLERemoteCharacteristic* pRabboniRemoteCharacteristic; BLEServer* pWebServer = NULL; BLECharacteristic* pWebTxCharacteristic = NULL; bool webDeviceConnected = false; // 系統運行狀態與感測數據 String systemMode = "manual"; // "auto" 或 "manual" float currentPitch = 0.0; float currentRoll = 0.0; unsigned long lastTelemetryTime = 0; const long telemetryInterval = 500; // 每 500ms 回傳一次狀態給網頁 const float TILT_THRESHOLD = 5.0; // 自動平衡啟動閾值 (度) // ==================== 3. Rabboni 數據通知回調 ==================== static void rabboniNotifyCallback( BLERemoteCharacteristic* pBLERemoteCharacteristic, uint8_t* pData, size_t length, bool isNotify) { if (length >= 12) { int16_t rawAx = (pData[1] << 8) | pData[0]; int16_t rawAy = (pData[3] << 8) | pData[2]; int16_t rawAz = (pData[5] << 8) | pData[4]; float ax = rawAx / 4096.0; float ay = rawAy / 4096.0; float az = rawAz / 4096.0; currentPitch = atan2(ax, sqrt(ay * ay + az * az)) * 180.0 / M_PI; currentRoll = atan2(ay, sqrt(ax * ax + az * az)) * 180.0 / M_PI; } } // ==================== 4. 連線 Rabboni 函數 ==================== bool connectToRabboni() { Serial.print("🔗 正在連線至 Rabboni 感測器: "); Serial.println(rabboniAddress.toString().c_str()); BLEClient* pClient = BLEDevice::createClient(); if (!pClient->connect(rabboniAddress)) { Serial.println("❌ 無法連線至 Rabboni"); return false; } Serial.println("✅ 已成功連線至 Rabboni!"); BLERemoteService* pRemoteService = pClient->getService("ffe0"); if (pRemoteService == nullptr) { pClient->disconnect(); return false; } // 💡 修正處：使用 getCharacteristic pRabboniRemoteCharacteristic = pRemoteService->getCharacteristic("ffe1"); if (pRabboniRemoteCharacteristic == nullptr) { pClient->disconnect(); return false; } if(pRabboniRemoteCharacteristic->canNotify()) { pRabboniRemoteCharacteristic->registerForNotify(rabboniNotifyCallback); } rabboniConnected = true; return true; } // ==================== 5. 網頁端藍牙事件回調 ==================== class MyServerCallbacks: public BLEServerCallbacks { void onConnected(BLEServer* pServer) { webDeviceConnected = true; Serial.println("🔗 網頁前端已連線"); }; void onDisconnected(BLEServer* pServer) { webDeviceConnected = false; Serial.println("🔌 網頁端已斷線，重新廣播..."); pServer->startAdvertising(); } }; class MyCallbacks: public BLECharacteristicCallbacks { void onWrite(BLECharacteristic *pCharacteristic) { String rxValue = pCharacteristic->getValue(); if (rxValue.length() > 0) { Serial.print("📥 收到指令: "); Serial.println(rxValue); if (rxValue.indexOf("auto") != -1) { systemMode = "auto"; Serial.println("⚙️ 切換至自動模式"); } else if (rxValue.indexOf("manual") != -1) { systemMode = "manual"; Serial.println("⚙️ 切換至手動模式"); } if (rxValue.indexOf("emergency_stop") != -1) { allStop(); return; } int firstComma = rxValue.indexOf(','); int secondComma = rxValue.indexOf(',', firstComma + 1); if (firstComma != -1 && secondComma != -1) { int col = rxValue.substring(0, firstComma).toInt(); String action = rxValue.substring(firstComma + 1, secondComma); int state = rxValue.substring(secondComma + 1).toInt(); if (systemMode == "manual") { controlActuator(col, action, state); } } } } }; // ==================== 6. 執行機構控制與安全保護 ==================== void controlActuator(int col, String action, int state) { if (col < 1 || col > 4) return; int pinIndex = -1; if (action == "pump") pinIndex = (col - 1) * 2; else if (action == "valve") pinIndex = (col - 1) * 2 + 1; if (pinIndex != -1) { digitalWrite(RELAY_PINS[pinIndex], state == 1 ? LOW : HIGH); // Active Low Serial.print("🕹️ 柱體 "); Serial.print(col); Serial.print(" ["); Serial.print(action); Serial.print("] -> "); Serial.println(state == 1 ? "ON" : "OFF"); } } void allStop() { for (int i = 0; i < 8; i++) digitalWrite(RELAY_PINS[i], HIGH); Serial.println("⚠️ 【安全保護】所有幫浦與閥門已強制關閉！"); } // ==================== 7. 自動平衡邏輯 ==================== void runAutoBalanceLogic() { if (currentPitch > TILT_THRESHOLD) { controlActuator(1, "pump", 1); } else if (currentPitch < -TILT_THRESHOLD) { controlActuator(3, "pump", 1); } else { for (int i = 0; i < 8; i++) digitalWrite(RELAY_PINS[i], HIGH); } } // ==================== 8. Setup ==================== void setup() { Serial.begin(115200); Serial.println("🚀 系統啟動中..."); for (int i = 0; i < 8; i++) { pinMode(RELAY_PINS[i], OUTPUT); digitalWrite(RELAY_PINS[i], HIGH); } BLEDevice::init("SemiSubmersible-ESP32"); pWebServer = BLEDevice::createServer(); pWebServer->setCallbacks(new MyServerCallbacks()); BLEService *pService = pWebServer->createService(WEB_SERVICE_UUID); pWebTxCharacteristic = pService->createCharacteristic(WEB_CHAR_TX_UUID, BLECharacteristic::PROPERTY_NOTIFY); pWebTxCharacteristic->addDescriptor(new BLE2902()); BLECharacteristic *pRxChar = pService->createCharacteristic(WEB_CHAR_RX_UUID, BLECharacteristic::PROPERTY_WRITE); pRxChar->setCallbacks(new MyCallbacks()); pService->start(); pWebServer->getAdvertising()->start(); connectToRabboni(); } // ==================== 9. Loop ==================== void loop() { if (!rabboniConnected && millis() % 5000 == 0) { connectToRabboni(); } if (systemMode == "auto" && rabboniConnected) { runAutoBalanceLogic(); } if (webDeviceConnected && (millis() - lastTelemetryTime >= telemetryInterval)) { lastTelemetryTime = millis(); String telemetryStr = "pitch:" + String(currentPitch, 2) + ",roll:" + String(currentRoll, 2) + ",mode:" + systemMode; pWebTxCharacteristic->setValue(telemetryStr.c_str()); pWebTxCharacteristic->notify(); } delay(50); }
